Schema جديد في Supabase

-- الكليات
CREATE TABLE faculties (
  id uuid PRIMARY KEY,
  name text NOT NULL,
  slug text UNIQUE,
  icon text,
  color text,
  description text
);

-- التخصصات
CREATE TABLE specialties (
  id uuid PRIMARY KEY,
  faculty_id uuid REFERENCES faculties(id),
  name text,
  years int -- عدد سنوات الدراسة
);

-- المواد
CREATE TABLE subjects (
  id uuid PRIMARY KEY,
  specialty_id uuid REFERENCES specialties(id),
  name text,
  year int, -- السنة الدراسية
  icon text
);

-- المواضيع
CREATE TABLE topics (
  id uuid PRIMARY KEY,
  subject_id uuid REFERENCES subjects(id),
  name text,
  order_index int
);

-- المستندات (محدّث)
ALTER TABLE documents
  ADD COLUMN specialty_id uuid REFERENCES specialties(id),
  ADD COLUMN subject_id uuid REFERENCES subjects(id),
  ADD COLUMN topic_id uuid REFERENCES topics(id),
  ADD COLUMN is_public boolean DEFAULT false,
  ADD COLUMN approval_status text DEFAULT 'pending',
  ADD COLUMN approved_by uuid REFERENCES auth.users(id),
  ADD COLUMN quality_score int,
  ADD COLUMN content_hash text; -- للمنع التكرار

Deduplication الذكي

-- عند رفع محاضرة جديدة:
-- 1. احسب embedding
-- 2. ابحث عن أقرب مستندات
-- 3. إذا التشابه > 90% → رفض
-- 4. إذا 70-90% → تحذير
-- 5. إذا < 70% → اقبل

CREATE OR REPLACE FUNCTION find_similar_documents(
  query_embedding vector(1024),
  filter_subject_id uuid,
  similarity_threshold float DEFAULT 0.85
)
RETURNS TABLE (
  doc_id uuid,
  similarity float,
  title text
) AS $$
BEGIN
  RETURN QUERY
  SELECT
    d.id,
    (1 - (dc.embedding <=> query_embedding))::float,
    d.filename
  FROM documents d
  JOIN document_chunks dc ON dc.document_id = d.id
  WHERE d.subject_id = filter_subject_id
    AND (1 - (dc.embedding <=> query_embedding)) > similarity_threshold
  ORDER BY dc.embedding <=> query_embedding
  LIMIT 5;
END;
$$ LANGUAGE plpgsql;

04 نظام التصميم

🎨 الاتجاه: Professional / Minimalist لا ألوان زاهية، لا أيقونات ملونة، لا gradients. تماماً مثل Linear و Vercel و Stripe. احترافي، نظيف، مركّز.

خلفيات داكنة متدرجة — أساس التصميم
ألوان الحالة فقط — تُستخدم بحذر شديد
